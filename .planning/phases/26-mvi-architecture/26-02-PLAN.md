    4. **Reducer 实现（纯函数）**：
    ```kotlin
    override fun reduce(state: DramaDetailUiState, intent: DramaDetailIntent): DramaDetailUiState {
        return when (intent) {
            // === 生命周期 ===
            is DramaDetailIntent.Init -> {
                performInitAsync()
                state.copy(initialSyncing = true)
            }
            is DramaDetailIntent.RetryInit -> {
                performInitAsync()
                state.copy(initialSyncing = true, initError = null)
            }

            // === WS 事件（per MVI-04）===
            is DramaDetailIntent.WsEventReceived -> handleWsEvent(state, intent.event)
            is DramaDetailIntent.ConnectionStateChanged -> state.copy(
                connectionState = if (intent.isConnected) ConnectionState.Connected else ConnectionState.Disconnected
            )

            // === 命令/聊天 ===
            is DramaDetailIntent.SendCommand -> {
                sendCommandAsync(intent.text)
                state.copy(isProcessing = true, isTyping = true)
            }
            is DramaDetailIntent.SendChatMessage -> {
                sendChatAsync(intent.text, intent.mention)
                state.copy(isProcessing = true)
            }

            // === 模式切换 ===
            is DramaDetailIntent.ToggleChatMode -> toggleChatMode(state)

            // === 场景历史 ===
            is DramaDetailIntent.LoadSceneHistory -> {
                loadSceneHistoryAsync()
                state.copy(showHistorySheet = true)
            }
            is DramaDetailIntent.ViewSceneHistory -> state.copy(viewingHistoryScene = intent.sceneNumber)
            is DramaDetailIntent.DismissSceneHistory -> state.copy(showHistorySheet = false, viewingHistoryScene = null)

            // === 保存 ===
            is DramaDetailIntent.ShowSaveDialog -> state.copy(showSaveDialog = true)
            is DramaDetailIntent.DismissSaveDialog -> state.copy(showSaveDialog = false)
            is DramaDetailIntent.SaveDrama -> {
                saveDramaAsync(intent.name)
                state.copy(showSaveDialog = false)
            }

            // === 导出 ===
            is DramaDetailIntent.ExportDrama -> {
                exportDramaAsync()
                state.copy(isExporting = true)
            }

            // === 演员面板 ===
            is DramaDetailIntent.ToggleActorDrawer -> state.copy(showActorDrawer = !state.showActorDrawer)
            is DramaDetailIntent.LoadActors -> {
                loadActorsAsync()
                state.copy(isActorLoading = true)
            }

            // === 打字超时 ===
            is DramaDetailIntent.TypingTimeout -> state.copy(
                isTyping = false, isProcessing = false, stormPhase = null,
                typingText = "AI 正在思考...", streamingActorName = "", streamingText = ""
            )

            // === 流式更新 ===
            is DramaDetailIntent.StreamingUpdate -> state.copy(
                streamingActorName = intent.actorName,
                streamingText = intent.text,
                isTyping = true
            )
            is DramaDetailIntent.StreamingComplete -> state.copy(
                streamingActorName = "", streamingText = ""
            )

            // === 内部异步结果 ===
            is DramaDetailIntent.Internal.InitSyncComplete -> state.copy(
                initialSyncing = false,
                initError = intent.error
            )
            is DramaDetailIntent.Internal.StatusLoaded -> state.withSyncedBubbles(intent.bubbles).copy(
                currentScene = intent.scene,
                tensionScore = intent.tension
            )
            is DramaDetailIntent.Internal.CommandResult -> {
                if (!intent.success) emitEffect(DramaDetailEffect.ShowSnackbar(intent.error ?: "命令失败"))
                state.copy(isProcessing = false, isTyping = false)
            }
            is DramaDetailIntent.Internal.ExportResult -> {
                emitEffect(DramaDetailEffect.ShareExport(intent.title, intent.content))
                state.copy(isExporting = false)
            }
            is DramaDetailIntent.Internal.ExportError -> {
                emitEffect(DramaDetailEffect.ShowSnackbar(intent.message))
                state.copy(isExporting = false)
            }
            is DramaDetailIntent.Internal.ActorsLoaded -> state.copy(
                actors = intent.actors,
                isActorLoading = false
            )
            is DramaDetailIntent.Internal.SceneHistoryLoaded -> state.copy(
                historyScenes = intent.scenes
            )
            is DramaDetailIntent.Internal.SaveComplete -> {
                emitEffect(DramaDetailEffect.ShowSnackbar("已保存：${intent.name}"))
                state
            }
            is DramaDetailIntent.Internal.LoadComplete -> {
                emitEffect(DramaDetailEffect.ShowSnackbar("已加载：${intent.name}"))
                state
            }
        }
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| IntentQueue → Reducer | 同进程串行，无需锁 |
| Reducer → State | 纯函数，原子赋值 |
| Internal Intent → 外部伪造 | sealed class 跨包限制 |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-26-03 | D (DoS) | 大量 WS 事件入队 | accept | UNLIMITED Channel + 轻量 Intent 对象，实际内存消耗极小；reduce 纯函数执行极快 |
| T-26-04 | E (Elevation) | Internal Intent 暴露 | mitigate | Internal 为 sealed class 子类，外部模块无法实例化（Kotlin 编译器限制） |
| T-26-05 | I (Disclosure) | Effect 重复消费 | accept | SharedFlow extraBufferCapacity=64，旧 Effect 自然丢弃；UI 层 LaunchedEffect 只收集当前存活周期内的 Effect |
</threat_model>
